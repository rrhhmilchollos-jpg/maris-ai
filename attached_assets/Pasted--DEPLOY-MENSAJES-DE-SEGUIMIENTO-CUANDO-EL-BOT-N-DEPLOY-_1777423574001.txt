const express = require('express');
const path = require('path');
const { execSync } = require('child_process');
const getPort = require('get-port');
const Project = require('../models/Project');
const authMiddleware = require('../middleware/auth');

const runningServers = new Map(); // projectId → { port, server }

router.post('/api/deploy/:projectId', authMiddleware, async (req, res) => {
  try {
    const { projectId } = req.params;
    const project = await Project.findOne({ 
      _id: projectId, 
      userId: req.user._id 
    });
    
    if (!project) return res.status(404).json({ error: 'Project not found' });

    const projectPath = path.join(process.env.PROJECTS_DIR, projectId);
    const frontendPath = path.join(projectPath, 'frontend');

    // Step 1: Install dependencies
    execSync('npm install', { 
      cwd: frontendPath, 
      timeout: 120000,
      stdio: 'pipe'
    });

    // Step 2: Build frontend
    execSync('npm run build', { 
      cwd: frontendPath, 
      timeout: 120000,
      stdio: 'pipe'
    });

    // Step 3: Find a free port
    const port = await getPort({ port: getPort.makeRange(4000, 5000) });

    // Step 4: Stop previous server if exists
    if (runningServers.has(projectId)) {
      runningServers.get(projectId).server.close();
    }

    // Step 5: Serve the built files
    const deployApp = express();
    deployApp.use(express.static(path.join(frontendPath, 'dist')));
    deployApp.get('*', (req, res) => {
      res.sendFile(path.join(frontendPath, 'dist', 'index.html'));
    });

    const deployServer = deployApp.listen(port);
    runningServers.set(projectId, { port, server: deployServer });

    const deploymentUrl = `http://localhost:${port}`;

    // Step 6: Save URL to MongoDB
    await Project.findByIdAndUpdate(projectId, { 
      deploymentUrl,
      status: 'deployed'
    });

    res.json({ 
      success: true, 
      deploymentUrl,
      port,
      message: 'App deployed successfully!'
    });

  } catch (error) {
    console.error('Deploy error:', error);
    res.status(500).json({ 
      error: 'Deploy failed', 
      details: error.message 
    });
  }
});

module.exports = router;

GitHub push is failing with authentication or API errors.

Fix the push logic completely in /backend/routes/github.js:

const { Octokit } = require('@octokit/rest');
const Project = require('../models/Project');
const User = require('../models/User');
const path = require('path');
const fs = require('fs').promises;
const authMiddleware = require('../middleware/auth');

router.post('/api/github/push/:projectId', authMiddleware, async (req, res) => {
  try {
    const user = await User.findById(req.user._id);
    
    if (!user.githubToken) {
      return res.status(401).json({ 
        error: 'GitHub not connected',
        connectUrl: '/api/auth/github'
      });
    }

    const project = await Project.findById(req.params.projectId);
    if (!project) return res.status(404).json({ error: 'Project not found' });

    const octokit = new Octokit({ auth: user.githubToken });
    
    // Get GitHub username
    const { data: githubUser } = await octokit.rest.users.getAuthenticated();
    const repoName = project.name
      .toLowerCase()
      .replace(/[^a-z0-9-]/g, '-')
      .replace(/-+/g, '-')
      .slice(0, 100);

    // Create repository
    let repo;
    try {
      const { data } = await octokit.rest.repos.createForAuthenticatedUser({
        name: repoName,
        description: `Built with AI on our platform`,
        private: false,
        auto_init: false
      });
      repo = data;
    } catch (err) {
      if (err.status === 422) {
        // Repo already exists, use it
        const { data } = await octokit.rest.repos.get({
          owner: githubUser.login,
          repo: repoName
        });
        repo = data;
      } else throw err;
    }

    // Push all project files
    const projectPath = path.join(process.env.PROJECTS_DIR, project._id.toString());
    
    async function getAllFiles(dirPath, baseDir = dirPath) {
      const files = [];
      const items = await fs.readdir(dirPath, { withFileTypes: true });
      
      for (const item of items) {
        const fullPath = path.join(dirPath, item.name);
        // Skip node_modules and dist
        if (item.name === 'node_modules' || item.name === 'dist' || item.name === '.git') continue;
        
        if (item.isDirectory()) {
          files.push(...await getAllFiles(fullPath, baseDir));
        } else {
          const relativePath = path.relative(baseDir, fullPath);
          const content = await fs.readFile(fullPath, 'utf-8');
          files.push({ path: relativePath, content });
        }
      }
      return files;
    }

    const allFiles = await getAllFiles(projectPath);

    // Push files one by one
    for (const file of allFiles) {
      const contentBase64 = Buffer.from(file.content).toString('base64');
      
      // Check if file exists (to get SHA for update)
      let sha;
      try {
        const { data } = await octokit.rest.repos.getContent({
          owner: githubUser.login,
          repo: repoName,
          path: file.path
        });
        sha = data.sha;
      } catch (e) {
        // File doesn't exist yet, that's fine
      }

      await octokit.rest.repos.createOrUpdateFileContents({
        owner: githubUser.login,
        repo: repoName,
        path: file.path,
        message: sha ? `Update ${file.path}` : `Add ${file.path}`,
        content: contentBase64,
        ...(sha ? { sha } : {})
      });
    }

    // Save repo URL to project
    await Project.findByIdAndUpdate(project._id, {
      githubRepo: repo.html_url
    });

    res.json({
      success: true,
      repoUrl: repo.html_url,
      repoName: repo.full_name,
      message: `Successfully pushed ${allFiles.length} files to GitHub`
    });

  } catch (error) {
    console.error('GitHub push error:', error);
    res.status(500).json({ 
      error: 'GitHub push failed',
      details: error.message
    });
  }
});

🔴 CUANDO EL REPO SE CREA PERO LOS ARCHIVOS NO SE SUBEN
GitHub repo is created successfully but files are not uploaded.
The repo appears empty.

This is a rate limiting issue with the GitHub API.
Fix by adding delays between file uploads and batching:

async function pushFilesWithRetry(octokit, owner, repo, files) {
  const BATCH_SIZE = 5;
  const DELAY_MS = 500;
  
  for (let i = 0; i < files.length; i += BATCH_SIZE) {
    const batch = files.slice(i, i + BATCH_SIZE);
    
    await Promise.all(batch.map(async (file) => {
      let retries = 3;
      while (retries > 0) {
        try {
          const contentBase64 = Buffer.from(file.content, 'utf-8').toString('base64');
          
          let sha;
          try {
            const { data } = await octokit.rest.repos.getContent({
              owner, repo: repo, path: file.path
            });
            sha = data.sha;
          } catch (e) {}

          await octokit.rest.repos.createOrUpdateFileContents({
            owner,
            repo,
            path: file.path,
            message: `Add ${file.path}`,
            content: contentBase64,
            ...(sha ? { sha } : {})
          });
          break; // Success, exit retry loop
        } catch (err) {
          retries--;
          if (retries === 0) throw err;
          await new Promise(r => setTimeout(r, 1000 * (4 - retries)));
        }
      }
    }));
    
    // Delay between batches to avoid rate limiting
    if (i + BATCH_SIZE < files.length) {
      await new Promise(r => setTimeout(r, DELAY_MS));
    }
  }
}
