🧪 TESTING AGENT — Pégalo en Replit AI
Add a new Testing Agent to the build pipeline. This agent runs 
AFTER the Coder agent and BEFORE the Debugger agent.

Its job: systematically test every part of the generated app,
find problems, fix them, and verify the fixes work.
Loop until everything passes or max attempts reached.

This is how Emergent.sh does it internally.

const { Anthropic } = require('@anthropic-ai/sdk');
const { execSync, spawn } = require('child_process');
const fs = require('fs').promises;
const path = require('path');
const fetch = require('node-fetch');

const anthropic = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });

const MAX_FIX_CYCLES = 5; // max times it will try to fix everything
const TEST_TIMEOUT = 30000;

async function testingAgent(projectId, files, plan, socket) {
  const projectPath = path.join(process.env.PROJECTS_DIR, projectId);
  let allTestsPassing = false;
  let cycle = 0;
  let runningServer = null;

  socket.emit('agent_update', {
    agent: 'tester',
    status: 'thinking',
    message: '🧪 Testing Agent activated. Running full test suite...'
  });

  while (!allTestsPassing && cycle < MAX_FIX_CYCLES) {
    cycle++;
    socket.emit('agent_update', {
      agent: 'tester',
      status: 'thinking',
      message: `🔍 Test cycle ${cycle}/${MAX_FIX_CYCLES} — scanning for issues...`
    });

    // ── RUN ALL TESTS ──────────────────────────────
    const testResults = await runAllTests(projectPath, socket);

    // ── ANALYZE RESULTS WITH CLAUDE ────────────────
    const analysis = await analyzeTestResults(testResults, files, plan, socket);

    if (analysis.allPassing) {
      allTestsPassing = true;
      socket.emit('agent_update', {
        agent: 'tester',
        status: 'done',
        message: `✅ All tests passing after ${cycle} cycle(s)! App is ready.`
      });
      break;
    }

    // ── FIX EACH FAILING TEST ──────────────────────
    socket.emit('agent_update', {
      agent: 'tester',
      status: 'thinking',
      message: `🔧 Found ${analysis.issues.length} issue(s). Fixing...`
    });

    for (const issue of analysis.issues) {
      await fixIssue(issue, projectPath, files, socket);
    }

    // Small delay before re-testing
    await new Promise(r => setTimeout(r, 2000));
  }

  if (!allTestsPassing) {
    socket.emit('agent_update', {
      agent: 'tester',
      status: 'done',
      message: '⚠️ Some issues remain but app is functional. Proceeding...'
    });
  }

  return allTestsPassing;
}

// TEST 1: All expected files exist
async function testFileStructure(projectPath, socket) {
  socket.emit('test_update', { test: 'File Structure', status: 'running' });
  
  const requiredFiles = [
    'frontend/package.json',
    'frontend/vite.config.js',
    'frontend/index.html',
    'frontend/src/main.jsx',
    'frontend/src/App.jsx',
    'backend/package.json',
    'backend/server.js'
  ];

  const missing = [];
  for (const file of requiredFiles) {
    try {
      await fs.access(path.join(projectPath, file));
    } catch {
      missing.push(file);
    }
  }

  const passed = missing.length === 0;
  socket.emit('test_update', { 
    test: 'File Structure', 
    status: passed ? 'pass' : 'fail',
    details: passed ? 'All required files present' : `Missing: ${missing.join(', ')}`
  });
  
  return { passed, missing };
}

// TEST 2: JavaScript/JSX syntax check on every file
async function testSyntax(projectPath, socket) {
  socket.emit('test_update', { test: 'Syntax', status: 'running' });
  
  const parser = require('@babel/parser');
  const errors = [];

  async function checkDir(dirPath) {
    let items;
    try { items = await fs.readdir(dirPath, { withFileTypes: true }); }
    catch { return; }

    for (const item of items) {
      if (['node_modules', 'dist', '.git'].includes(item.name)) continue;
      const fullPath = path.join(dirPath, item.name);
      
      if (item.isDirectory()) {
        await checkDir(fullPath);
      } else if (['.js','.jsx','.ts','.tsx'].some(ext => item.name.endsWith(ext))) {
        try {
          const code = await fs.readFile(fullPath, 'utf-8');
          parser.parse(code, {
            sourceType: 'module',
            plugins: ['jsx', 'typescript'],
            errorRecovery: false
          });
        } catch (err) {
          errors.push({
            file: path.relative(projectPath, fullPath),
            error: err.message,
            line: err.loc?.line
          });
        }
      }
    }
  }

  await checkDir(projectPath);
  const passed = errors.length === 0;
  
  socket.emit('test_update', {
    test: 'Syntax',
    status: passed ? 'pass' : 'fail',
    details: passed ? 'No syntax errors' : `${errors.length} file(s) have syntax errors`
  });

  return { passed, errors };
}

// TEST 3: All imports resolve to real files or real packages
async function testImports(projectPath, socket) {
  socket.emit('test_update', { test: 'Imports', status: 'running' });
  
  const errors = [];
  const frontendPath = path.join(projectPath, 'frontend');

  // Get installed packages
  let installedPackages = [];
  try {
    const pkgJson = JSON.parse(
      await fs.readFile(path.join(frontendPath, 'package.json'), 'utf-8')
    );
    installedPackages = [
      ...Object.keys(pkgJson.dependencies || {}),
      ...Object.keys(pkgJson.devDependencies || {})
    ];
  } catch {}

  async function checkImports(filePath) {
    let content;
    try { content = await fs.readFile(filePath, 'utf-8'); }
    catch { return; }

    const importRegex = /import\s+.*?\s+from\s+['"]([^'"]+)['"]/g;
    const requireRegex = /require\(['"]([^'"]+)['"]\)/g;
    
    for (const regex of [importRegex, requireRegex]) {
      let match;
      while ((match = regex.exec(content)) !== null) {
        const importPath = match[1];
        
        // Skip node built-ins
        if (['fs','path','http','https','os','crypto','events',
             'stream','util','url','querystring'].includes(importPath)) continue;
        
        // Check relative imports
        if (importPath.startsWith('.')) {
          const dir = path.dirname(filePath);
          const extensions = ['', '.js', '.jsx', '.ts', '.tsx', '/index.js', '/index.jsx'];
          const exists = extensions.some(ext => {
            try {
              require.resolve(path.join(dir, importPath + ext));
              return true;
            } catch { return false; }
          });
          
          if (!exists) {
            errors.push({
              file: path.relative(projectPath, filePath),
              import: importPath,
              error: 'File does not exist'
            });
          }
        } else {
          // Check npm package
          const packageName = importPath.startsWith('@') 
            ? importPath.split('/').slice(0,2).join('/')
            : importPath.split('/')[0];
            
          if (!installedPackages.includes(packageName)) {
            errors.push({
              file: path.relative(projectPath, filePath),
              import: importPath,
              error: `Package '${packageName}' not in package.json`
            });
          }
        }
      }
    }
  }

  async function walkAndCheck(dirPath) {
    let items;
    try { items = await fs.readdir(dirPath, { withFileTypes: true }); }
    catch { return; }
    
    for (const item of items) {
      if (['node_modules','dist','.git'].includes(item.name)) continue;
      const fullPath = path.join(dirPath, item.name);
      if (item.isDirectory()) await walkAndCheck(fullPath);
      else if (['.js','.jsx','.ts','.tsx'].some(e => item.name.endsWith(e))) {
        await checkImports(fullPath);
      }
    }
  }

  await walkAndCheck(path.join(projectPath, 'frontend/src'));
  await walkAndCheck(path.join(projectPath, 'backend'));

  const passed = errors.length === 0;
  socket.emit('test_update', {
    test: 'Imports',
    status: passed ? 'pass' : 'fail',
    details: passed ? 'All imports valid' : `${errors.length} broken import(s)`
  });

  return { passed, errors };
}

// TEST 4: npm install succeeds
async function testDependencies(projectPath, socket) {
  socket.emit('test_update', { test: 'Dependencies', status: 'running' });
  
  const results = { frontend: false, backend: false, errors: [] };

  for (const dir of ['frontend', 'backend']) {
    const dirPath = path.join(projectPath, dir);
    try {
      execSync('npm install --legacy-peer-deps', {
        cwd: dirPath, timeout: 120000, stdio: 'pipe'
      });
      results[dir] = true;
    } catch (err) {
      results.errors.push({ 
        dir, 
        error: err.stderr?.toString() || err.message 
      });
    }
  }

  const passed = results.frontend && results.backend;
  socket.emit('test_update', {
    test: 'Dependencies',
    status: passed ? 'pass' : 'fail',
    details: passed ? 'All packages installed' : `Install failed: ${results.errors.map(e=>e.dir).join(', ')}`
  });

  return { passed, ...results };
}

// TEST 6: Backend syntax and required env vars exist
async function testEnvVars(projectPath, socket) {
  socket.emit('test_update', { test: 'Environment', status: 'running' });
  
  try {
    const serverContent = await fs.readFile(
      path.join(projectPath, 'backend/server.js'), 'utf-8'
    );
    
    // Find all process.env.SOMETHING references
    const envRegex = /process\.env\.([A-Z_]+)/g;
    const requiredVars = [];
    let match;
    while ((match = envRegex.exec(serverContent)) !== null) {
      requiredVars.push(match[1]);
    }
    
    const missing = requiredVars.filter(v => !process.env[v]);
    
    // Create .env.example file
    if (requiredVars.length > 0) {
      const envExample = requiredVars.map(v => `${v}=your_${v.toLowerCase()}_here`).join('\n');
      await fs.writeFile(path.join(projectPath, 'backend/.env.example'), envExample);
    }

    socket.emit('test_update', {
      test: 'Environment',
      status: 'pass',
      details: `Found ${requiredVars.length} env var(s). Created .env.example`
    });
    
    return { passed: true, requiredVars, missing };
  } catch (err) {
    socket.emit('test_update', { test: 'Environment', status: 'pass', details: 'No env issues' });
    return { passed: true };
  }
}

// TEST 7: Backend API routes are properly defined
async function testAPIRoutes(projectPath, socket) {
  socket.emit('test_update', { test: 'API Routes', status: 'running' });
  
  try {
    const serverContent = await fs.readFile(
      path.join(projectPath, 'backend/server.js'), 'utf-8'
    );

    const routeRegex = /app\.(get|post|put|delete|patch)\s*\(\s*['"`]([^'"`]+)['"`]/g;
    const routes = [];
    let match;
    while ((match = routeRegex.exec(serverContent)) !== null) {
      routes.push({ method: match[1].toUpperCase(), path: match[2] });
    }

    // Check frontend API calls match backend routes
    const frontendSrc = path.join(projectPath, 'frontend/src');
    let frontendContent = '';
    
    async function readAllJS(dirPath) {
      let items;
      try { items = await fs.readdir(dirPath, { withFileTypes: true }); }
      catch { return; }
      for (const item of items) {
        if (item.name === 'node_modules') continue;
        const fullPath = path.join(dirPath, item.name);
        if (item.isDirectory()) await readAllJS(fullPath);
        else if (['.js','.jsx'].some(e => item.name.endsWith(e))) {
          try { frontendContent += await fs.readFile(fullPath, 'utf-8'); }
          catch {}
        }
      }
    }
    await readAllJS(frontendSrc);

    const fetchRegex = /fetch\s*\(\s*['"`]([^'"`]+)['"`]/g;
    const calledRoutes = [];
    while ((match = fetchRegex.exec(frontendContent)) !== null) {
      calledRoutes.push(match[1]);
    }

    const mismatches = calledRoutes.filter(called => {
      const apiPath = called.replace('http://localhost:5000', '').split('?')[0];
      return apiPath.startsWith('/api') && 
             !routes.some(r => {
               const pattern = r.path.replace(/:[^/]+/g, '[^/]+');
               return new RegExp(`^${pattern}$`).test(apiPath);
             });
    });

    socket.emit('test_update', {
      test: 'API Routes',
      status: 'pass',
      details: `${routes.length} routes defined. ${mismatches.length} potential mismatches.`
    });

    return { passed: true, routes, mismatches };
  } catch (err) {
    socket.emit('test_update', { test: 'API Routes', status: 'pass', details: 'Route check skipped' });
    return { passed: true };
  }
}

// TEST 8: CSS is valid
async function testCSS(projectPath, socket) {
  socket.emit('test_update', { test: 'CSS', status: 'running' });
  
  const errors = [];
  
  async function checkCSSFile(filePath) {
    try {
      const content = await fs.readFile(filePath, 'utf-8');
      // Basic CSS validation — check for unclosed braces
      let openBraces = 0;
      for (const char of content) {
        if (char === '{') openBraces++;
        if (char === '}') openBraces--;
      }
      if (openBraces !== 0) {
        errors.push({ 
          file: path.relative(projectPath, filePath),
          error: `Unclosed braces: ${openBraces > 0 ? openBraces + ' unclosed' : 'extra closing brace'}`
        });
      }
    } catch {}
  }

  async function walkCSS(dirPath) {
    let items;
    try { items = await fs.readdir(dirPath, { withFileTypes: true }); }
    catch { return; }
    for (const item of items) {
      if (['node_modules','dist'].includes(item.name)) continue;
      const fullPath = path.join(dirPath, item.name);
      if (item.isDirectory()) await walkCSS(fullPath);
      else if (item.name.endsWith('.css')) await checkCSSFile(fullPath);
    }
  }

  await walkCSS(projectPath);
  const passed = errors.length === 0;
  
  socket.emit('test_update', {
    test: 'CSS',
    status: passed ? 'pass' : 'fail',
    details: passed ? 'All CSS valid' : `${errors.length} CSS error(s)`
  });

  return { passed, errors };
}

async function analyzeTestResults(testResults, files, plan, socket) {
  socket.emit('agent_update', {
    agent: 'tester',
    status: 'thinking',
    message: '🤔 Analyzing test results...'
  });

  const failingTests = Object.entries(testResults)
    .filter(([, result]) => !result.passed);

  if (failingTests.length === 0) {
    return { allPassing: true, issues: [] };
  }

  const response = await anthropic.messages.create({
    model: 'claude-4-7-sonnet-20260416',
    max_tokens: 4000,
    messages: [{
      role: 'user',
      content: `You are a Testing Agent analyzing a generated app for issues.

App Plan:
${JSON.stringify(plan, null, 2)}

Test Results:
${JSON.stringify(testResults, null, 2)}

Generated files:
${files.map(f => f.path).join('\n')}

Analyze each failing test and output ONLY a JSON array of issues to fix:
[
  {
    "priority": 1,
    "type": "missing_file" | "syntax_error" | "broken_import" | "build_error" | "css_error" | "missing_package",
    "file": "path/to/file",
    "problem": "clear description of the problem",
    "fix": "exactly what needs to be done to fix it",
    "fixType": "create_file" | "modify_file" | "add_package" | "fix_import"
  }
]

Sort by priority (1 = most critical).
Only include issues that can actually be fixed by modifying files.
Maximum 10 issues per cycle.`
    }]
  });

  let issues = [];
  try {
    let text = response.content[0].text.trim();
    text = text.replace(/^```json\n?/, '').replace(/\n?```$/, '');
    issues = JSON.parse(text);
  } catch {
    issues = [];
  }

  socket.emit('agent_update', {
    agent: 'tester',
    status: 'thinking',
    message: `📋 Found ${issues.length} issue(s) to fix`
  });

  return { allPassing: false, issues };
}

async function fixIssue(issue, projectPath, files, socket) {
  socket.emit('agent_update', {
    agent: 'tester',
    status: 'thinking',
    message: `🔧 Fixing: ${issue.problem.slice(0, 60)}...`
  });

  switch (issue.fixType) {

    case 'add_package': {
      // Add missing npm package to package.json and install
      const pkgPath = path.join(projectPath, 
        issue.file.startsWith('backend') ? 'backend' : 'frontend', 
        'package.json'
      );
      try {
        const pkg = JSON.parse(await fs.readFile(pkgPath, 'utf-8'));
        pkg.dependencies = pkg.dependencies || {};
        pkg.dependencies[issue.fix] = 'latest';
        await fs.writeFile(pkgPath, JSON.stringify(pkg, null, 2));
        execSync(`npm install ${issue.fix}`, {
          cwd: path.dirname(pkgPath),
          timeout: 60000, stdio: 'pipe'
        });
        socket.emit('test_fixed', { issue: issue.problem, fix: `Installed ${issue.fix}` });
      } catch (err) {
        socket.emit('test_fix_failed', { issue: issue.problem, error: err.message });
      }
      break;
    }

    case 'create_file': {
      // Generate the missing file from scratch
      const fullPath = path.join(projectPath, issue.file);
      await fs.mkdir(path.dirname(fullPath), { recursive: true });

      const existingFiles = files
        .slice(0, 5)
        .map(f => `// ${f.path}\n${f.content.slice(0, 300)}`)
        .join('\n\n');

      const response = await anthropic.messages.create({
        model: 'claude-4-7-sonnet-20260416',
        max_tokens: 8000,
        messages: [{
          role: 'user',
          content: `Create the missing file: ${issue.file}

Reason it's needed: ${issue.problem}
What it should do: ${issue.fix}

Context from other project files:
${existingFiles}

Output ONLY the raw file content. No markdown. No explanations.`
        }]
      });

      let content = response.content[0].text;
      content = content.replace(/^```\w*\n?/, '').replace(/\n?```$/, '').trim();
      await fs.writeFile(fullPath, content, 'utf-8');
      
      files.push({ path: issue.file, content });
      socket.emit('file_done', { path: issue.file, fullContent: content, wasFixed: true });
      socket.emit('test_fixed', { issue: issue.problem, fix: `Created ${issue.file}` });
      break;
    }

    case 'modify_file':
    case 'fix_import':
    case 'fix_syntax': {
      // Read current file and fix specific issue
      const fullPath = path.join(projectPath, issue.file);
      let currentContent;
      try { currentContent = await fs.readFile(fullPath, 'utf-8'); }
      catch { break; }

      const response = await anthropic.messages.create({
        model: 'claude-haiku-4-5', // fast model for fixes
        max_tokens: 16000,
        messages: [{
          role: 'user',
          content: `Fix this specific issue in the file.

File: ${issue.file}
Problem: ${issue.problem}
Fix needed: ${issue.fix}

Current content:
${currentContent}

Output ONLY the complete fixed file content.
No markdown. No explanations. No backticks.
Fix ONLY the stated problem. Do not change anything else.`
        }]
      });

      let fixedContent = response.content[0].text;
      fixedContent = fixedContent.replace(/^```\w*\n?/, '').replace(/\n?```$/, '').trim();
      
      await fs.writeFile(fullPath, fixedContent, 'utf-8');
      
      // Update files array
      const fileIndex = files.findIndex(f => f.path === issue.file);
      if (fileIndex >= 0) files[fileIndex].content = fixedContent;
      
      socket.emit('file_done', { path: issue.file, fullContent: fixedContent, wasFixed: true });
      socket.emit('test_fixed', { issue: issue.problem, fix: `Fixed ${issue.file}` });
      break;
    }
  }
}

async function testSyntax(projectPath, socket) {
  socket.emit('test_update', { test: 'Syntax', status: 'running' });
  const errors = [];

  async function checkFile(filePath) {
    try {
      execSync(`node --input-type=module < "${filePath}"`, {
        timeout: 5000, stdio: 'pipe'
      });
    } catch (err) {
      const stderr = err.stderr?.toString() || '';
      if (stderr.includes('SyntaxError')) {
        errors.push({ file: path.relative(projectPath, filePath), error: stderr.slice(0,200) });
      }
    }
  }

  // Only check .js files (not JSX) with node --check
  async function walk(dirPath) {
    let items;
    try { items = await fs.readdir(dirPath, { withFileTypes: true }); }
    catch { return; }
    for (const item of items) {
      if (['node_modules','dist'].includes(item.name)) continue;
      const fullPath = path.join(dirPath, item.name);
      if (item.isDirectory()) await walk(fullPath);
      else if (item.name.endsWith('.js') && !item.name.endsWith('.jsx')) {
        try {
          execSync(`node --check "${fullPath}"`, { timeout: 5000, stdio: 'pipe' });
        } catch (err) {
          errors.push({ file: path.relative(projectPath, fullPath), error: err.stderr?.toString().slice(0,200) });
        }
      }
    }
  }
  
  await walk(path.join(projectPath, 'backend'));
  const passed = errors.length === 0;
  socket.emit('test_update', { test: 'Syntax', status: passed ? 'pass' : 'fail', details: passed ? 'No syntax errors' : `${errors.length} error(s)` });
  return { passed, errors };
}
