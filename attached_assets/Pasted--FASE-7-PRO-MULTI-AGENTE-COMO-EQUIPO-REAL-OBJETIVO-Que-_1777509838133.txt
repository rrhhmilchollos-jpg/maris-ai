🧠 2. NUEVO ORQUESTADOR (TEAM MODE)
/orchestrator/teamOrchestrator.js
const { runAgent } = require("../agents/router");

async function runTeam(goal) {
  console.log("🚀 Iniciando equipo para:", goal);

  const plan = await runAgent({
    type: "architect",
    input: goal
  });

  const tasks = plan.tasks;

  const results = await Promise.all(
    tasks.map(task => runAgent(task))
  );

  return results;
}

module.exports = { runTeam };
🧠 3. NUEVO AGENTE: ARCHITECT (EL JEFE)

Este es el más importante.

/agents/architect.js
const OpenAI = require("openai");
const client = new OpenAI({ apiKey: process.env.OPENAI_KEY });

async function architect(goal) {
  const prompt = `
Eres un arquitecto de software senior.

Divide este proyecto en módulos reales:

${goal}

Responde en JSON:
{
  "tasks": [
    { "type": "backend", "input": "..." },
    { "type": "frontend", "input": "..." },
    { "type": "qa", "input": "..." }
  ]
}
`;

  const res = await client.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [{ role: "user", content: prompt }]
  });

  return JSON.parse(res.choices[0].message.content);
}

module.exports = { architect };
🤖 4. BACKEND / FRONTEND EN PARALELO

Ejemplo router:

async function runAgent(task) {
  switch (task.type) {
    case "backend":
      return backendAgent(task.input);

    case "frontend":
      return frontendAgent(task.input);

    case "qa":
      return qaAgent(task.input);

    case "review":
      return reviewerAgent(task.input);

    case "devops":
      return devopsAgent(task.input);
  }
}
