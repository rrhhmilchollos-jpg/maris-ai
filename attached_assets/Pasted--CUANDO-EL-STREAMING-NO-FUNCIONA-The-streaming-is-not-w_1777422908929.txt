🔴 CUANDO LOS AGENTES NO SE EJECUTAN EN SECUENCIA
The agents are all running at the same time or skipping steps.

Fix the pipeline in /backend/routes/builder.js to be strictly sequential:

async function runBuildPipeline(socket, projectId, prompt, model) {
  try {
    // STEP 1
    socket.emit('agent_update', { agent: 'planner', status: 'thinking', message: 'Analyzing your request...' });
    const plan = await plannerAgent(prompt);
    socket.emit('agent_update', { agent: 'planner', status: 'done', message: 'Architecture planned!' });

    // STEP 2
    socket.emit('agent_update', { agent: 'coder', status: 'thinking', message: 'Writing your code...' });
    const files = await coderAgent(plan, model, socket);
    socket.emit('agent_update', { agent: 'coder', status: 'done', message: 'Code complete!' });

    // STEP 3
    socket.emit('agent_update', { agent: 'debugger', status: 'thinking', message: 'Running health check...' });
    const isHealthy = await debuggerAgent(projectId, files, socket);
    socket.emit('agent_update', { agent: 'debugger', status: 'done', message: isHealthy ? 'No issues found!' : 'Fixed all errors!' });

    // STEP 4
    socket.emit('agent_update', { agent: 'designer', status: 'thinking', message: 'Polishing the UI...' });
    await designerAgent(projectId, files, socket);
    socket.emit('agent_update', { agent: 'designer', status: 'done', message: 'Design improved!' });

    // DONE
    socket.emit('build_complete', { projectId, previewUrl: `/preview/${projectId}`, files });

  } catch (error) {
    socket.emit('build_error', { message: error.message, recoverable: false });
  }
}

Each agent function must be async and must fully resolve before the next one starts.
Use await — do not use .then() chains.

🔴 CUANDO EL FOLLOW-UP CHAT NO MODIFICA EL APP
Follow-up messages are not modifying the existing app.
They either rebuild from scratch or do nothing.

Fix the follow-up handler in /backend/routes/builder.js:

socket.on('follow_up', async ({ message, projectId }) => {
  // 1. Load current project files from MongoDB
  const project = await Project.findById(projectId);
  const currentFiles = project.files;

  // 2. Mini planner call — identify which files to change
  const plannerResponse = await anthropic.messages.create({
    model: 'claude-opus-4-6',
    max_tokens: 1000,
    messages: [{
      role: 'user',
      content: `The user wants to modify their existing app.
      Change requested: "${message}"
      Current files: ${currentFiles.map(f => f.path).join(', ')}
      
      Return ONLY a JSON array of file paths that need to change:
      ["frontend/src/App.jsx", "frontend/src/styles.css"]`
    }]
  });
  
  const filesToChange = JSON.parse(plannerResponse.content[0].text);

  // 3. For each file that needs changing, rewrite it
  for (const filePath of filesToChange) {
    const currentFile = currentFiles.find(f => f.path === filePath);
    
    socket.emit('agent_update', { 
      agent: 'coder', 
      status: 'thinking', 
      message: `Modifying ${filePath}...` 
    });

    const stream = await anthropic.messages.stream({
      model: 'claude-sonnet-4-6',
      max_tokens: 8192,
      messages: [{
        role: 'user',
        content: `Modify this file based on the request: "${message}"
        
        Current content:
        ${currentFile.content}
        
        Return ONLY the complete new file content, no explanations.`
      }]
    });

    let newContent = '';
    for await (const chunk of stream) {
      if (chunk.type === 'content_block_delta') {
        newContent += chunk.delta.text;
        socket.emit('file_chunk', { path: filePath, chunk: chunk.delta.text });
      }
    }

    // 4. Write to disk and update MongoDB
    const fullPath = path.join(PROJECTS_DIR, projectId, filePath);
    await fs.writeFile(fullPath, newContent);
    
    await Project.updateOne(
      { _id: projectId, 'files.path': filePath },
      { $set: { 'files.$.content': newContent } }
    );

    socket.emit('file_done', { path: filePath, fullContent: newContent });
  }

  // 5. Rebuild frontend and refresh preview
  execSync('npm run build', { cwd: path.join(PROJECTS_DIR, projectId, 'frontend') });
  socket.emit('build_complete', { projectId, refreshPreview: true });
});

const deductCredits = async (userId, amount) => {
  const user = await User.findById(userId);
  if (user.credits.remaining < amount) {
    throw new Error('INSUFFICIENT_CREDITS');
