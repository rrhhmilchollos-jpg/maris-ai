const fs = require("fs");
const path = require("path");

function createProject(name) {
  const projectPath = path.join(process.cwd(), "projects", name);

  fs.mkdirSync(projectPath, { recursive: true });

  return projectPath;
}

module.exports = { createProject };

🔧 4. TOOL GIT (VERSIONADO REAL)
/tools/gitTool.js
const { exec } = require("child_process");

function run(cmd, cwd) {
  return new Promise((res, rej) => {
    exec(cmd, { cwd }, (err, stdout) => {
      if (err) return rej(err);
      res(stdout);
    });
  });
}

async function initRepo(path) {
  await run("git init", path);
  await run("git add .", path);
  await run('git commit -m "init"', path);
}

module.exports = { initRepo };
