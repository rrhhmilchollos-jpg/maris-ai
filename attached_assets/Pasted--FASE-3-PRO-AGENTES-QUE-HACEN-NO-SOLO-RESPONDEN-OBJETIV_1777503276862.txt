📁 2. TOOL: CREAR Y EDITAR ARCHIVOS
/tools/fileTool.js
const fs = require("fs");
const path = require("path");

function writeFile(filePath, content) {
  const fullPath = path.join(process.cwd(), filePath);

  fs.mkdirSync(path.dirname(fullPath), { recursive: true });
  fs.writeFileSync(fullPath, content);

  return `Archivo creado: ${filePath}`;
}

function readFile(filePath) {
  return fs.readFileSync(path.join(process.cwd(), filePath), "utf-8");
}

module.exports = { writeFile, readFile };
💻 3. TOOL: EJECUTAR COMANDOS
/tools/execTool.js
const { exec } = require("child_process");

function runCommand(command) {
  return new Promise((resolve, reject) => {
    exec(command, (err, stdout, stderr) => {
      if (err) return reject(stderr);
      resolve(stdout);
    });
  });
}

module.exports = { runCommand };
🧠 4. AGENTE BUILDER PRO (GENERA CÓDIGO REAL)
/agents/builder.js
const OpenAI = require("openai");
const { writeFile } = require("../tools/fileTool");

const client = new OpenAI({ apiKey: process.env.OPENAI_KEY });

async function builder(input) {
  const prompt = `
Crea código funcional basado en esto:

${input}

Devuelve JSON:
{
  "file": "ruta/archivo.js",
  "content": "codigo completo"
}
`;

  const res = await client.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [{ role: "user", content: prompt }]
  });

  const data = JSON.parse(res.choices[0].message.content);

  await writeFile(data.file, data.content);

  return `✅ Archivo creado: ${data.file}`;
}

module.exports = { builder };
🔧 5. AGENTE FIXER PRO (AUTOCORRECCIÓN REAL)
/agents/fixer.js
const OpenAI = require("openai");
const { readFile, writeFile } = require("../tools/fileTool");

const client = new OpenAI({ apiKey: process.env.OPENAI_KEY });

async function fixer(input) {
  const file = "app.js"; // ejemplo base

  const code = readFile(file);

  const prompt = `
Corrige este código:

${code}

Error:
${input}

Devuelve código corregido completo
`;

  const res = await client.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [{ role: "user", content: prompt }]
  });

  const fixedCode = res.choices[0].message.content;

  await writeFile(file, fixedCode);

  return "🔧 Código corregido";
}

module.exports = { fixer };
