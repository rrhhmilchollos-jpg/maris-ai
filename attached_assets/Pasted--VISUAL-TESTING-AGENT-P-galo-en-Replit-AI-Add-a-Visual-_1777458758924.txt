const puppeteer = require('puppeteer');
const sharp = require('sharp');
const fs = require('fs').promises;
const path = require('path');
const { Anthropic } = require('@anthropic-ai/sdk');

const anthropic = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });

const MAX_VISUAL_FIX_CYCLES = 3;

async function visualTestingAgent(projectId, files, plan, previewUrl, socket) {
  socket.emit('agent_update', {
    agent: 'visual_tester',
    status: 'thinking',
    message: '👁️ Visual Testing Agent activated. Taking screenshots...'
  });

  let browser;
  let cycle = 0;
  let visuallyPassing = false;

  try {
    browser = await puppeteer.launch({
      headless: 'new',
      args: [
        '--no-sandbox',
        '--disable-setuid-sandbox',
        '--disable-dev-shm-usage',
        '--disable-gpu',
        '--window-size=1280,800'
      ]
    });

    while (!visuallyPassing && cycle < MAX_VISUAL_FIX_CYCLES) {
      cycle++;

      socket.emit('agent_update', {
        agent: 'visual_tester',
        status: 'thinking',
        message: `📸 Visual check cycle ${cycle}/${MAX_VISUAL_FIX_CYCLES}...`
      });

      // ── TAKE SCREENSHOTS ──────────────────────────
      const screenshots = await takeScreenshots(browser, previewUrl, projectId, socket);

      if (!screenshots || screenshots.length === 0) {
        socket.emit('agent_update', {
          agent: 'visual_tester',
          status: 'done',
          message: '⚠️ Could not take screenshots. Skipping visual check.'
        });
        break;
      }

      // ── ANALYZE WITH CLAUDE VISION ────────────────
      const analysis = await analyzeScreenshots(screenshots, plan, socket);

      // ── EMIT SCREENSHOTS TO FRONTEND ──────────────
      socket.emit('visual_screenshots', {
        screenshots: screenshots.map(s => ({
          viewport: s.viewport,
          base64: s.base64,
          url: s.url
        }))
      });

      if (analysis.visuallyCorrect) {
        visuallyPassing = true;
        socket.emit('agent_update', {
          agent: 'visual_tester',
          status: 'done',
          message: `✅ Visual tests passing! App looks great.`
        });
        break;
      }

      // ── FIX VISUAL ISSUES ─────────────────────────
      socket.emit('agent_update', {
        agent: 'visual_tester',
        status: 'thinking',
        message: `🎨 Found ${analysis.issues.length} visual issue(s). Fixing...`
      });

      for (const issue of analysis.issues) {
        await fixVisualIssue(issue, projectId, files, plan, socket);
      }

      // Rebuild frontend after fixes
      const { execSync } = require('child_process');
      try {
        execSync('npm run build', {
          cwd: path.join(process.env.PROJECTS_DIR, projectId, 'frontend'),
          timeout: 120000,
          stdio: 'pipe'
        });
        socket.emit('agent_update', {
          agent: 'visual_tester',
          status: 'thinking',
          message: '🔄 Rebuilt after visual fixes. Re-checking...'
        });
      } catch (err) {
        socket.emit('agent_update', {
          agent: 'visual_tester',
          status: 'done',
          message: '⚠️ Rebuild failed after visual fixes. Using previous build.'
        });
        break;
      }

      // Wait for server to restart
      await new Promise(r => setTimeout(r, 3000));
    }

  } catch (err) {
    socket.emit('agent_update', {
      agent: 'visual_tester',
      status: 'done',
      message: '⚠️ Visual testing encountered an error. Proceeding...'
    });
    console.error('Visual tester error:', err);
  } finally {
    if (browser) await browser.close();
  }

  return visuallyPassing;
}

STEP 3 — SCREENSHOT FUNCTION (multiple viewports)
=======================================================

async function takeScreenshots(browser, previewUrl, projectId, socket) {
  const viewports = [
    { name: 'desktop', width: 1280, height: 800 },
    { name: 'tablet',  width: 768,  height: 1024 },
    { name: 'mobile',  width: 390,  height: 844 }
  ];

  const screenshots = [];
  const screenshotDir = path.join(
    process.env.PROJECTS_DIR, projectId, 'screenshots'
  );
  await fs.mkdir(screenshotDir, { recursive: true });

  const page = await browser.newPage();

  // Capture console errors from the app
  const consoleErrors = [];
  page.on('console', msg => {
    if (msg.type() === 'error') consoleErrors.push(msg.text());
  });
  page.on('pageerror', err => consoleErrors.push(err.message));

  for (const viewport of viewports) {
    try {
      await page.setViewport({ 
        width: viewport.width, 
        height: viewport.height,
        deviceScaleFactor: 1
      });

      socket.emit('visual_test_update', {
        viewport: viewport.name,
        status: 'capturing',
        message: `📸 Capturing ${viewport.name} (${viewport.width}x${viewport.height})...`
      });

      // Navigate to the app
      await page.goto(previewUrl, {
        waitUntil: 'networkidle2',
        timeout: 30000
      });

      // Wait for React to render
      await page.waitForTimeout(2000);

      // Scroll to capture lazy content
      await page.evaluate(() => window.scrollTo(0, 300));
      await page.waitForTimeout(500);
      await page.evaluate(() => window.scrollTo(0, 0));
      await page.waitForTimeout(500);

      // Take full page screenshot
      const screenshotBuffer = await page.screenshot({
        fullPage: true,
        type: 'png'
      });

      // Resize for Claude vision (max 1568px wide)
      const resizedBuffer = await sharp(screenshotBuffer)
        .resize({ width: Math.min(viewport.width, 1568), withoutEnlargement: true })
        .png({ quality: 85 })
        .toBuffer();

      const base64 = resizedBuffer.toString('base64');
      const filePath = path.join(screenshotDir, `${viewport.name}.png`);
      await fs.writeFile(filePath, resizedBuffer);

      screenshots.push({
        viewport: viewport.name,
        width: viewport.width,
        height: viewport.height,
        base64,
        filePath,
        consoleErrors: [...consoleErrors]
      });

      socket.emit('visual_test_update', {
        viewport: viewport.name,
        status: 'captured',
        message: `✅ ${viewport.name} screenshot captured`
      });

      consoleErrors.length = 0; // reset for next viewport

    } catch (err) {
      socket.emit('visual_test_update', {
        viewport: viewport.name,
        status: 'failed',
        message: `❌ Could not capture ${viewport.name}: ${err.message}`
      });
    }
  }

  await page.close();
  return screenshots;
}

=======================================================
STEP 4 — CLAUDE VISION ANALYSIS
=======================================================

async function analyzeScreenshots(screenshots, plan, socket) {
  socket.emit('agent_update', {
    agent: 'visual_tester',
    status: 'thinking',
    message: '🧠 Claude is analyzing the screenshots...'
  });

  // Build content array with all screenshots
  const content = [];

  for (const screenshot of screenshots) {
    content.push({
      type: 'text',
      text: `=== ${screenshot.viewport.toUpperCase()} VIEWPORT (${screenshot.width}x${screenshot.height}) ===`
    });
    content.push({
      type: 'image',
      source: {
        type: 'base64',
        media_type: 'image/png',
        data: screenshot.base64
      }
    });
    if (screenshot.consoleErrors.length > 0) {
      content.push({
        type: 'text',
        text: `Console errors on this viewport:\n${screenshot.consoleErrors.join('\n')}`
      });
    }
  }

  content.push({
    type: 'text',
    text: `
App Plan:
- Name: ${plan.appName}
- Type: ${plan.appType}
- Features: ${plan.features?.join(', ')}
- Expected color scheme: ${JSON.stringify(plan.colorScheme)}

Analyze these screenshots and identify visual problems.
Output ONLY valid JSON with this exact structure:
{
  "visuallyCorrect": boolean,
  "overallScore": number (0-100),
  "issues": [
    {
      "severity": "critical" | "major" | "minor",
      "type": "blank_page" | "broken_layout" | "missing_content" | 
              "bad_colors" | "not_responsive" | "overlapping_elements" | 
              "missing_navbar" | "console_errors" | "ugly_design" |
              "wrong_font" | "missing_images" | "broken_links_visual",
      "viewport": "desktop" | "tablet" | "mobile" | "all",
      "description": "specific description of what is wrong",
      "affectedFile": "frontend/src/App.jsx or relevant CSS file",
      "cssfix": "the exact CSS or JSX change needed to fix this"
    }
  ],
  "positives": ["what looks good"],
  "summary": "one sentence overall assessment"
}

Be strict. A good app should score 80+.
Flag: blank pages, broken layouts, missing key features,
bad responsive design, ugly colors, overlapping elements,
missing navigation, console errors visible on screen.
Do NOT flag minor aesthetic preferences.`
  });

  const response = await anthropic.messages.create({
    model: 'claude-sonnet-4-6',
    max_tokens: 4000,
    messages: [{ role: 'user', content }]
  });

  let analysis;
  try {
    let text = response.content[0].text.trim();
    text = text.replace(/^```json\n?/, '').replace(/\n?```$/, '').trim();
    analysis = JSON.parse(text);
  } catch {
    analysis = { visuallyCorrect: true, issues: [], overallScore: 70, summary: 'Analysis parse failed, assuming OK' };
  }

  socket.emit('visual_analysis', {
    score: analysis.overallScore,
    summary: analysis.summary,
    issueCount: analysis.issues?.length || 0,
    positives: analysis.positives || []
  });

  socket.emit('agent_update', {
    agent: 'visual_tester',
    status: 'thinking',
    message: `📊 Visual score: ${analysis.overallScore}/100 — ${analysis.summary}`
  });

  // Only fix critical and major issues
  analysis.issues = (analysis.issues || [])
    .filter(i => i.severity !== 'minor')
    .slice(0, 5); // max 5 fixes per cycle

  analysis.visuallyCorrect = 
    analysis.overallScore >= 70 && 
    !analysis.issues.some(i => i.severity === 'critical');

  return analysis;
}

STEP 5 — AUTO-FIX VISUAL ISSUES

async function fixVisualIssue(issue, projectId, files, plan, socket) {
  socket.emit('agent_update', {
    agent: 'visual_tester',
    status: 'thinking',
    message: `🎨 Fixing visual issue: ${issue.description.slice(0, 60)}...`
  });

  const projectPath = path.join(process.env.PROJECTS_DIR, projectId);

  // Special handler for blank page
  if (issue.type === 'blank_page') {
    await fixBlankPage(projectId, files, plan, socket);
    return;
  }

  // Special handler for console errors
  if (issue.type === 'console_errors') {
    await fixConsoleErrors(projectId, files, socket);
    return;
  }

  // For layout/CSS issues, use Claude to fix the specific file
  const filePath = issue.affectedFile || 'frontend/src/App.jsx';
  const fullPath = path.join(projectPath, filePath);

  let currentContent;
  try {
    currentContent = await fs.readFile(fullPath, 'utf-8');
  } catch {
    socket.emit('visual_test_update', {
      viewport: issue.viewport,
      status: 'failed',
      message: `File not found: ${filePath}`
    });
    return;
  }

  const response = await anthropic.messages.create({
    model: 'claude-sonnet-4-6',
    max_tokens: 16000,
    messages: [{
      role: 'user',
      content: `Fix this visual issue in the app.

Issue type: ${issue.type}
Severity: ${issue.severity}
Viewport affected: ${issue.viewport}
Description: ${issue.description}
Suggested fix: ${issue.cssfix}

File to fix: ${filePath}
Current content:
${currentContent}

App color scheme: ${JSON.stringify(plan.colorScheme)}
App type: ${plan.appType}

Rules:
- Fix ONLY the described visual issue
- Do not change functionality
- Do not change other working styles
- Make it look professional and modern
- Ensure the fix works on all viewports
- If fixing responsive issues, use proper Tailwind breakpoints 
  (sm: md: lg:) or CSS media queries
- Output ONLY the complete fixed file content
- No markdown, no backticks, no explanations`
    }]
  });

  let fixedContent = response.content[0].text;
  fixedContent = fixedContent.replace(/^```\w*\n?/, '').replace(/\n?```$/, '').trim();

  await fs.writeFile(fullPath, fixedContent, 'utf-8');

  const fileIndex = files.findIndex(f => f.path === filePath);
  if (fileIndex >= 0) files[fileIndex].content = fixedContent;
  else files.push({ path: filePath, content: fixedContent });

  socket.emit('file_done', {
    path: filePath,
    fullContent: fixedContent,
    wasFixed: true,
    fixType: 'visual'
  });

  socket.emit('visual_test_update', {
    viewport: issue.viewport,
    status: 'fixed',
    message: `✅ Fixed: ${issue.description.slice(0, 60)}`
  });
}

// Fix blank page specifically
async function fixBlankPage(projectId, files, plan, socket) {
  socket.emit('agent_update', {
    agent: 'visual_tester',
    status: 'thinking',
    message: '🚨 Blank page detected. Regenerating main component...'
  });

  const projectPath = path.join(process.env.PROJECTS_DIR, projectId);
  const appPath = path.join(projectPath, 'frontend/src/App.jsx');
  const mainPath = path.join(projectPath, 'frontend/src/main.jsx');

  // Fix main.jsx first
  const mainContent = `
import React from 'react'
import ReactDOM from 'react-dom/client'
import App from './App.jsx'
import './index.css'

ReactDOM.createRoot(document.getElementById('root')).render(
  <React.StrictMode>
    <App />
  </React.StrictMode>,
)
`.trim();

  await fs.writeFile(mainPath, mainContent, 'utf-8');

  // Regenerate App.jsx with guaranteed visible content
  const response = await anthropic.messages.create({
    model: 'claude-sonnet-4-6',
    max_tokens: 16000,
    messages: [{
      role: 'user',
      content: `The app is showing a blank page. 
      
Regenerate App.jsx for this app:
- Name: ${plan.appName}
- Type: ${plan.appType}  
- Features: ${plan.features?.join(', ')}
- Colors: ${JSON.stringify(plan.colorScheme)}

Requirements:
- Must render visible content immediately (no loading states that never resolve)
- Must have a proper navbar with the app name
- Must show the main feature of the app
- Use inline Tailwind classes for styling
- Background color: ${plan.colorScheme?.background || '#0a0a0f'}
- Primary color: ${plan.colorScheme?.primary || '#7c3aed'}
- Text color: ${plan.colorScheme?.text || '#f8fafc'}
- Must be a complete working React component
- Export default App

Output ONLY the raw JSX code. No markdown.`
    }]
  });

  let appContent = response.content[0].text;
  appContent = appContent.replace(/^```\w*\n?/, '').replace(/\n?```$/, '').trim();

  await fs.writeFile(appPath, appContent, 'utf-8');

  const appIndex = files.findIndex(f => f.path === 'frontend/src/App.jsx');
  if (appIndex >= 0) files[appIndex].content = appContent;

  socket.emit('file_done', {
    path: 'frontend/src/App.jsx',
    fullContent: appContent,
    wasFixed: true,
    fixType: 'blank_page'
  });
}

// Fix console errors visible in screenshots
async function fixConsoleErrors(projectId, files, socket) {
  const { execSync } = require('child_process');
  const projectPath = path.join(process.env.PROJECTS_DIR, projectId);

  try {
    // Re-run the tester's syntax and import checks
    execSync('node --check server.js', {
      cwd: path.join(projectPath, 'backend'),
      stdio: 'pipe',
      timeout: 10000
    });
  } catch (err) {
    const errorText = err.stderr?.toString() || err.message;
    socket.emit('agent_update', {
      agent: 'visual_tester',
      status: 'thinking',
      message: `🔧 Fixing console error: ${errorText.slice(0, 80)}...`
    });
  }
}

async function runBuildPipeline(socket, { prompt, projectId, model, userId }) {

  // STEP 1: Planner
  socket.emit('agent_update', { agent: 'planner', status: 'thinking', message: 'Planning your app...' });
  const plan = await plannerAgent(prompt);
  socket.emit('agent_update', { agent: 'planner', status: 'done', message: 'Architecture ready!' });

  // STEP 2: Coder
  socket.emit('agent_update', { agent: 'coder', status: 'thinking', message: 'Writing your code...' });
  const files = await coderAgent(plan, model, socket, projectId);
  socket.emit('agent_update', { agent: 'coder', status: 'done', message: 'Code complete!' });

  // STEP 3: Testing Agent
  socket.emit('agent_update', { agent: 'tester', status: 'thinking', message: 'Running test suite...' });
  await testingAgent(projectId, files, plan, socket);

  // STEP 4: Deploy preview to get a URL for screenshots
  const previewUrl = await deployPreview(projectId);

  // STEP 5: Visual Testing Agent ← NEW
  socket.emit('agent_update', { agent: 'visual_tester', status: 'thinking', message: 'Visual testing...' });
  await visualTestingAgent(projectId, files, plan, previewUrl, socket);

  // STEP 6: Designer
  socket.emit('agent_update', { agent: 'designer', status: 'thinking', message: 'Final polish...' });
  await designerAgent(projectId, files, socket);
  socket.emit('agent_update', { agent: 'designer', status: 'done', message: 'Design complete!' });

  // STEP 7: Save and complete
  await Project.findByIdAndUpdate(projectId, { status: 'ready', files });

  socket.emit('build_complete', {
    projectId,
    previewUrl,
    files
  });
}

// Helper: deploy preview before visual testing
async function deployPreview(projectId) {
  const { execSync } = require('child_process');
  const getPort = require('get-port');
  const express = require('express');
  const path = require('path');

  const frontendPath = path.join(process.env.PROJECTS_DIR, projectId, 'frontend');
  
  execSync('npm run build', { cwd: frontendPath, timeout: 120000, stdio: 'pipe' });
  
  const port = await getPort({ port: getPort.makeRange(4000, 5000) });
  const app = require('express')();
  app.use(require('express').static(path.join(frontendPath, 'dist')));
  app.get('*', (req, res) => res.sendFile(path.join(frontendPath, 'dist/index.html')));
  app.listen(port, '0.0.0.0');
  
  await new Promise(r => setTimeout(r, 2000));
  return `http://localhost:${port}`;
}
