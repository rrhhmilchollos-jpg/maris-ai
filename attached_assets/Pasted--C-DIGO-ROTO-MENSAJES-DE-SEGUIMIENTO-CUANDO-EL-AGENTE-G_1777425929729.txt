async function generateFileWithContinuation(prompt, socket, filePath) {
  let fullContent = '';
  let isComplete = false;
  let iteration = 0;
  const MAX_ITERATIONS = 5;

  while (!isComplete && iteration < MAX_ITERATIONS) {
    const messages = iteration === 0
      ? [{ role: 'user', content: prompt }]
      : [
          { role: 'user', content: prompt },
          { role: 'assistant', content: fullContent },
          { role: 'user', content: 'Continue exactly from where you left off. Do not repeat anything.' }
        ];

    const stream = await anthropic.messages.stream({
      model: 'claude-sonnet-4-6',
      max_tokens: 16000,
      messages
    });

    let chunkContent = '';
    for await (const chunk of stream) {
      if (chunk.type === 'content_block_delta') {
        chunkContent += chunk.delta.text;
        socket.emit('file_chunk', { path: filePath, chunk: chunk.delta.text });
      }
    }

    fullContent += chunkContent;

    // Check if file looks complete
    const trimmed = fullContent.trim();
    isComplete = (
      trimmed.endsWith('}') ||
      trimmed.endsWith('};') ||
      trimmed.endsWith('/>') ||
      trimmed.endsWith('</div>') ||
      trimmed.endsWith('```')
    );

    iteration++;
  }

  return fullContent;
}

const parser = require('@babel/parser');

function validateJSX(code, filePath) {
  if (!filePath.endsWith('.jsx') && !filePath.endsWith('.js') && !filePath.endsWith('.tsx')) {
    return { valid: true };
  }
  
  try {
    parser.parse(code, {
      sourceType: 'module',
      plugins: ['jsx', 'typescript'],
      errorRecovery: false
    });
    return { valid: true };
  } catch (err) {
    return { 
      valid: false, 
      error: err.message,
      line: err.loc?.line,
      column: err.loc?.column
    };
  }
}

function extractCodeFromResponse(response, filePath) {
  let code = response.trim();
  
  // Remove markdown code blocks with language specifier
  // e.g. ```jsx, ```javascript, ```css, ```json
  const codeBlockRegex = /^```(?:jsx?|tsx?|javascript|typescript|css|json|html|bash|sh)?\n([\s\S]*?)```$/;
  const match = code.match(codeBlockRegex);
  
  if (match) {
    code = match[1].trim();
  }
  
  // Also handle code blocks without language
  const genericBlockRegex = /^```\n?([\s\S]*?)```$/;
  const genericMatch = code.match(genericBlockRegex);
  if (genericMatch) {
    code = genericMatch[1].trim();
  }
  
  // Remove any leading/trailing backticks
  code = code.replace(/^`+|`+$/g, '').trim();
  
  return code;
}

async function debuggerAgent(projectId, errors, socket) {
  const MAX_RETRIES = 3;
  const fixedFiles = new Set();
  
  for (let attempt = 1; attempt <= MAX_RETRIES; attempt++) {
    socket.emit('agent_update', {
      agent: 'debugger',
      status: 'thinking',
      message: `Fix attempt ${attempt}/${MAX_RETRIES}...`
    });

    // Group errors by file
    const errorsByFile = {};
    for (const err of errors) {
      if (!errorsByFile[err.file]) errorsByFile[err.file] = [];
      errorsByFile[err.file].push(err.error);
    }

    let anyFixed = false;

    for (const [filePath, fileErrors] of Object.entries(errorsByFile)) {
      // Skip files we already tried fixing twice
      const fixKey = `${filePath}-${attempt}`;
      if (fixedFiles.has(fixKey)) continue;
      fixedFiles.add(fixKey);

      const fullPath = path.join(process.env.PROJECTS_DIR, projectId, filePath);
      
      let currentContent;
      try {
        currentContent = await fs.readFile(fullPath, 'utf-8');
      } catch (e) {
        continue; // File doesn't exist, skip
      }

      const response = await anthropic.messages.create({
        model: 'claude-haiku-4-5',
        max_tokens: 16000,
        messages: [{
          role: 'user',
          content: `Fix these errors in this file.

File: ${filePath}
Errors:
${fileErrors.join('\n')}

Current file content:
${currentContent}

Rules:
- Return ONLY the complete fixed file content
- Fix ALL the listed errors
- Do not introduce new errors
- Do not change working functionality
- If the error is an import that doesn't exist, remove that import and inline the code instead`
        }]
      });

      let fixedContent = response.content[0].text;
      fixedContent = extractCodeFromResponse(fixedContent, filePath);

      await fs.writeFile(fullPath, fixedContent, 'utf-8');
      
      socket.emit('file_done', { 
        path: filePath, 
        fullContent: fixedContent,
        wasFixed: true,
        attempt
      });

      anyFixed = true;
    }

    // Re-run health check after fixes
    const { healthy, errors: newErrors } = await runHealthCheck(projectId, socket);
    
    if (healthy) {
      socket.emit('agent_update', {
        agent: 'debugger',
        status: 'done',
        message: `✅ All errors fixed after ${attempt} attempt(s)!`
      });
      return true;
    }

    if (!anyFixed) {
      // Nothing was fixed this round, no point retrying
      break;
    }

    errors = newErrors; // Use new errors for next iteration
  }

  // Max retries reached — do not loop more
  socket.emit('agent_update', {
    agent: 'debugger',
    status: 'done',
    message: '⚠️ Some issues remain. App may still work partially.'
  });

  // Still emit build_complete so user can see what was built
  socket.emit('build_complete', {
    projectId,
    previewUrl: `/preview/${projectId}`,
    hasWarnings: true,
    message: 'Built with warnings. Some features may not work perfectly.'
  });

  return false;
}

const BUILD_TIMEOUT_MS = 5 * 60 * 1000; // 5 minutes max

async function runBuildPipeline(socket, data) {
  const timeoutPromise = new Promise((_, reject) => {
    setTimeout(() => {
      reject(Object.assign(
        new Error('Build timed out after 5 minutes'),
        { phase: 'timeout', recoverable: true }
      ));
    }, BUILD_TIMEOUT_MS);
  });

  // Send heartbeat every 10 seconds so frontend knows we're alive
  const heartbeat = setInterval(() => {
    socket.emit('heartbeat', { timestamp: Date.now() });
  }, 10000);

  try {
    await Promise.race([
      runAgentPipeline(socket, data),
      timeoutPromise
    ]);
  } finally {
    clearInterval(heartbeat);
  }
}

async function verifyPreviewIsActuallyWorking(previewUrl, maxAttempts = 10) {
  const delay = (ms) => new Promise(r => setTimeout(r, ms));
  
  for (let i = 1; i <= maxAttempts; i++) {
    try {
      const response = await fetch(previewUrl, { 
        method: 'GET',
        timeout: 5000,
        signal: AbortSignal.timeout(5000)
      });
      
      if (response.ok) {
        const html = await response.text();
        // Make sure it's not an empty page
        if (html.length > 100 && html.includes('<html')) {
          return { working: true, attempt: i };
        }
      }
    } catch (err) {
      // Not ready yet, keep trying
    }
    
    // Wait before retry (exponential backoff)
    await delay(Math.min(1000 * i, 5000));
  }
  
  return { working: false };
}
