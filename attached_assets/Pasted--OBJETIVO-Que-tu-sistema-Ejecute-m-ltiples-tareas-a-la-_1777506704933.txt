🔧 2. COLA DE TRABAJOS (EL CORAZÓN)

Usa BullMQ + Redis

/queue/queue.js
const { Queue } = require("bullmq");

const queue = new Queue("tasks", {
  connection: {
    host: "127.0.0.1",
    port: 6379
  }
});

async function addJob(data) {
  await queue.add("task", data);
}

module.exports = { addJob };
⚙️ 3. WORKERS (EJECUCIÓN REAL EN PARALELO)
/workers/worker.js
const { Worker } = require("bullmq");
const { runAgent } = require("../agents/router");

const worker = new Worker(
  "tasks",
  async job => {
    const { task } = job.data;

    console.log("⚡ Ejecutando:", task);

    const result = await runAgent(task);

    return result;
  },
  {
    concurrency: 5, // 🔥 clave: 5 tareas simultáneas
    connection: {
      host: "127.0.0.1",
      port: 6379
    }
  }
);

📡 5. TRACKING DE RESULTADOS

Necesitas saber qué pasa con cada tarea.

/queue/events.js
const { QueueEvents } = require("bullmq");

const queueEvents = new QueueEvents("tasks");

queueEvents.on("completed", ({ jobId, returnvalue }) => {
  console.log(`✅ Job ${jobId} completado`, returnvalue);
});

queueEvents.on("failed", ({ jobId, failedReason }) => {
  console.log(`❌ Job ${jobId} falló`, failedReason);
});
