/memory/longTerm.js
const { Pool } = require("pg");

const pool = new Pool({
  connectionString: process.env.DB_URL
});

async function saveMemory(goal, result, success) {
  await pool.query(
    "INSERT INTO memory(goal, result, success) VALUES($1,$2,$3)",
    [goal, result, success]
  );
}

async function getSimilarMemories(goal) {
  const res = await pool.query(
    "SELECT * FROM memory ORDER BY created_at DESC LIMIT 5"
  );
  return res.rows;
}

module.exports = { saveMemory, getSimilarMemories };

🧠 2. MEMORY MANAGER (EL CEREBRO)
/memory/memoryManager.js
const { saveMemory, getSimilarMemories } = require("./longTerm");

async function enrichContext(goal) {
  const past = await getSimilarMemories(goal);

  return `
Contexto previo:
${past.map(p => `- ${p.goal} → ${p.result}`).join("\n")}
`;
}

module.exports = { enrichContext, saveMemory };

async function cacheResult(key, value) {
  await redis.set(key, JSON.stringify(value));
}

  const data = await redis.get(key);
  return data ? JSON.parse(data) : null;
}
