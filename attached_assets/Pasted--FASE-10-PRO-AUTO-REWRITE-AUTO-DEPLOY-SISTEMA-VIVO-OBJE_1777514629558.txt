/evolution/codeAnalyzer.js
const fs = require("fs");

function analyzeCodebase() {
  const files = fs.readdirSync("./agents");

  return files.map(file => {
    const content = fs.readFileSync(`./agents/${file}`, "utf-8");

    return {
      file,
      size: content.length,
      hasErrors: content.includes("undefined")
    };
  });
}

module.exports = { analyzeCodebase };

👉 el sistema ahora “lee su propio cerebro”

🧠 3. CODE WRITER (CREA AGENTES NUEVOS)
/evolution/codeWriter.js
const fs = require("fs");
const OpenAI = require("openai");

const client = new OpenAI({ apiKey: process.env.OPENAI_KEY });

async function generateAgent(spec) {
  const prompt = `
Crea un agente de IA en JavaScript.

Especificación:
${spec}

Debe ser funcional y exportar una función async.
`;

  const res = await client.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [{ role: "user", content: prompt }]
  });

  const code = res.choices[0].message.content;

  const fileName = `agent_${Date.now()}.js`;

  fs.writeFileSync(`./agents/${fileName}`, code);

  return fileName;
}

module.exports = { generateAgent };

👉 ahora el sistema crea agentes nuevos SOLO

const { analyzeCodebase } = require("./codeAnalyzer");
const { generateAgent } = require("./codeWriter");
const { testCode } = require("./sandboxTester");
const { deploy } = require("./deployManager");

async function evolutionLoop() {
  const report = analyzeCodebase();

  if (report.some(f => f.hasErrors)) {
    const newAgent = await generateAgent("Fix system errors automatically");

    const test = await testCode(newAgent);

    if (test.success) {
      await deploy();
    }
  }
}

setInterval(evolutionLoop, 30000);
